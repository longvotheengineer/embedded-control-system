 module xuat_xung(clk, PulseOut, mode);
	input 	clk;
	input	[1:0] mode;
	output	PulseOut;
	reg		PulseOut;
	reg		[31:0] clk_cnt, T;
	
	initial begin
		PulseOut	= 0;
		clk_cnt		= 0;
		T 		 	= 0;
	end
	
	always @(posedge clk) begin
		case (mode)
			2'b00: T = 1000 - 1;
			2'b01: T = 5000 - 1;
			2'b10: T = 2000 - 1;
			2'b11: T = 10000000 - 1;
		endcase
		
		if (clk_cnt < T) begin
			clk_cnt  <= clk_cnt + 1;
			PulseOut <= 0;
		end
		else begin
			clk_cnt  <= 0;
			PulseOut <= 1;
		end
	end
 endmodule

 #define ADDR_PWM	0x60000040
 #define SIZE_BUFF	35
 
 volatile uint16_t	*ptr_pwm;
		  uint16_t	pwm_val;
		  uint8_t	rxbuff[SIZE_BUFF];
 
 bool check_CRC (uint8_t *rxbuff) {
	uint8_t crc = 0;
	for (int i = 0; i < 34; i++) {
		crc ^= rxbuff[i];
	}
	
	if (rxbuff[ 0] == 0xAA &&
		rxbuff[ 1] == 0x55 &&
		rxbuff[34] == crc) {
		return true;
	}
	else 	
		return false;
 }
 
 int main() {
	while(1) {
		receive_data ((char *)rxbuff, SIZE_BUFF);
		ptr_pwm = (volatile uint16_t *)ADDR_PWM;
		
		if (check_CRC (rxbuff)) {					
			for (int i = 1; i <= 16; i++) {
				pwm_val  	= (rxbuff[2*i] << 8) + rxbuff[2*i + 1];
				*ptr_pwm++ 	= pwm_val;
			}
		}
	}
 }
